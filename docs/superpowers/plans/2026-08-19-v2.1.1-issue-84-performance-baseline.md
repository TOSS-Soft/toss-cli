Create `scripts/performance/report.mjs`:

```js
import {canonicalJson} from "../../src/contracts/acp.js";

export const PERFORMANCE_REPORT_VERSION="toss-test-performance-report.v1";
export const PERFORMANCE_BASELINE_VERSION="toss-test-performance-baseline.v1";
export const HISTORICAL_FULL_WALL_MS=134960;
export const FAST_MAX_WALL_MS=15000;

function finiteNonnegative(value,label) {
  if (!Number.isFinite(value) || value<0) {
    throw new TypeError(`${label} must be finite nonnegative`);
  }
  return value;
}

export function median(values) {
  if (!Array.isArray(values) || values.length!==3) {
    throw new TypeError("median requires exactly three samples");
  }
  return [...values]
    .map((value,index) => finiteNonnegative(value,`sample ${index+1}`))
    .sort((left,right) => left-right)[1];
}

function ascii(left,right) {
  return left<right ? -1 : left>right ? 1 : 0;
}

export function parseNamedDurations(output) {
  if (typeof output!=="string") throw new TypeError("output must be a string");
  const rows=[];
  for (const line of output.split(/\r?\n/u)) {
    const match=/^([✔✖])\s+(.+?)\s+\((\d+(?:\.\d+)?)ms\)$/u.exec(line.trim());
    if (!match) continue;
    rows.push({
      name:match[2],duration_ms:Number(match[3]),
      status:match[1]==="✔" ? "pass" : "fail",
    });
  }
  return rows.sort((left,right) => ascii(left.name,right.name) || left.duration_ms-right.duration_ms);
}
```

```js
import {appendFileSync} from "node:fs";

const logPath=process.env.TOSS_PERFORMANCE_PROCESS_LOG;

if (logPath && runId) {
```

Create `scripts/performance/run-suite.mjs` with these exact boundaries:

```js
import {spawn} from "node:child_process";
import {mkdtemp,readFile,rm,writeFile} from "node:fs/promises";
import {tmpdir} from "node:os";
import {join} from "node:path";
import {performance} from "node:perf_hooks";
import {fileURLToPath,pathToFileURL} from "node:url";

import {parseNamedDurations,summarizeProcessEvents} from "./report.mjs";

const probe=fileURLToPath(new URL("./process-probe.mjs",import.meta.url));

function preloadOptions(existing="") {
  const own=`--import=${pathToFileURL(probe).href}`;
  return existing.trim()==="" ? own : `${existing} ${own}`;
}

function execute(command,args,{cwd,env}) {
  return new Promise((resolveExecution,rejectExecution) => {
    const child=spawn(command,args,{cwd,env,stdio:["ignore","pipe","pipe"],shell:false});
    let stdout="";
    let stderr="";
    child.stdout.setEncoding("utf8");
    child.stderr.setEncoding("utf8");
    child.stdout.on("data",chunk => { stdout+=chunk; });
    child.stderr.on("data",chunk => { stderr+=chunk; });
    child.once("error",rejectExecution);
    child.once("close",status => resolveExecution({status:status ?? 70,stdout,stderr}));
  });
}

export async function runSuiteOnce({command,args,cwd,runId,env={}}) {
  const scratch=await mkdtemp(join(tmpdir(),"toss-test-performance-"));
  const log=join(scratch,"processes.jsonl");
  await writeFile(log,"","utf8");
  const started=performance.now();
  try {
    const result=await execute(command,args,{
      cwd,
      env:{
        ...process.env,...env,
        NODE_OPTIONS:preloadOptions(env.NODE_OPTIONS ?? process.env.NODE_OPTIONS ?? ""),
        TOSS_PERFORMANCE_PROCESS_LOG:log,
        TOSS_PERFORMANCE_RUN_ID:runId,
      },
    });
    const events=(await readFile(log,"utf8")).split(/\r?\n/u)
      .filter(Boolean).map(line => JSON.parse(line));
    const processes=summarizeProcessEvents(events,cwd,runId);
    return {
      wall_ms:performance.now()-started,
      user_cpu_ms:processes.user_cpu_ms,
      system_cpu_ms:processes.system_cpu_ms,
      exit_status:result.status,
      fresh_process_count:processes.fresh_process_count,
      peak_process_count:processes.peak_process_count,
      duplicates:processes.duplicates,
      slowest_files:processes.entries.slice(0,10),
      slowest_tests:parseNamedDurations(result.stdout)
        .sort((left,right) => right.duration_ms-left.duration_ms).slice(0,10),
      stdout:result.stdout,stderr:result.stderr,
    };
  } finally {
    await rm(scratch,{recursive:true,force:true});
  }
}
```

Cap stdout and stderr at 64 MiB each. On overflow, terminate the child and return status 70 with `PERFORMANCE_OUTPUT_LIMIT` in stderr. Reject malformed JSONL, mixed run IDs, duplicate starts/ends, and incomplete pairs with the Task 1 stable error codes. Count Node processes whose entry point is outside the repository, such as npm itself, in CPU and process totals but omit them from test-file and duplicate-entry analysis.

```js
export async function runBenchmark({runs,lane,runnerId,cwd,identity,runOnce=runSuiteOnce}) {
  if (runs!==3) throw new TypeError("benchmark requires exactly three runs");
  const exactIdentity=identity ?? await collectIdentity(cwd,runnerId);
  const npm=process.platform==="win32" ? "npm.cmd" : "npm";
  const samples=[];
  for (let index=0;index<3;index+=1) {
    samples.push(await runOnce({
      command:npm,args:["test"],cwd,
      runId:`${exactIdentity.commit}-${index+1}`,env:{},
    }));
  }
  return createPerformanceReport({lane,identity:exactIdentity,samples});
}

export function createBaseline(report) {
  if (report.lane!=="full") throw new TypeError("baseline requires the full lane");
  const basis=Math.min(HISTORICAL_FULL_WALL_MS,report.medians.wall_ms);
  return Object.freeze({
    schema_version:PERFORMANCE_BASELINE_VERSION,
    identity:report.identity,
    historical:{full_wall_ms:HISTORICAL_FULL_WALL_MS},
    samples:report.samples,medians:report.medians,
    budgets:{
      fast_max_wall_ms:FAST_MAX_WALL_MS,
      full_max_wall_ms:Math.floor(basis*0.70),
    },
  });
}

export function renderBenchmarkOutput(report,json) {
  if (json) return {stdout:canonicalPerformanceJson(report),stderr:""};
  return {
    stdout:[
      `${report.lane} median: ${report.medians.wall_ms}ms`,
      `user CPU: ${report.medians.user_cpu_ms}ms`,
      `system CPU: ${report.medians.system_cpu_ms}ms`,
      `processes: ${report.medians.fresh_process_count}`,
    ].join("\n")+"\n",
    stderr:"",
  };
}
```

Only `--update-baseline` calls `createBaseline`.
