# TOSS CLI v2.1.1 Issue #85 Explicit Test Lanes Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace duplicate, unbounded test discovery with one versioned ownership manifest and a deterministic cross-platform lane runner while preserving every existing correctness, package, workflow, documentation, and release check.

**Architecture:** A checked-in JSON manifest owns every executable `scripts/*-test.js` and top-level `test/*.test.js` entry exactly once in one of five ownership lanes. A pure validator compares that manifest with a direct eligibility scan plus a recursive unexpected-test audit and fails closed before execution. A Node.js runner selects a lane or the stable full union, schedules explicit paths with a checked-in concurrency of one, buffers each entry result for deterministic output, and preserves every child status. Package scripts, benchmarks, CI, `prepack`, and release validation all route through that same runner without shell globs.

**Tech Stack:** Node.js 20+ ESM, built-in `node:test`, `node:assert/strict`, `node:child_process`, filesystem APIs, JSON, YAML, npm scripts, GitHub Actions, and Markdown.

**Spec:** `docs/superpowers/specs/2026-08-19-v2.1.1-test-efficiency-design.md`

**Issue:** [#85](https://github.com/TOSS-Soft/toss-cli/issues/85)

```js
export const TEST_MANIFEST_VERSION="toss-test-manifest.v1";
export const OWNERSHIP_LANES=Object.freeze([
  "fast","integration","e2e","package","release",
]);
export const REQUESTED_LANES=Object.freeze([...OWNERSHIP_LANES,"full"]);

export function selectTestEntries(manifest,lane) {
  if (!REQUESTED_LANES.includes(lane)) {
    throw new TypeError(`unknown test lane: ${lane}`);
  }
  return lane==="full"

}
```

**Interfaces:**
- `executeTestEntry(entry,{cwd,env}) -> Promise<EntryResult>`
- `runTestLane({lane,cwd,manifest,eligibleEntries,executeEntry,stdout,stderr}) -> Promise<LaneResult>`
- `EntryResult` is `{entry,outcome,exit_status,signal,error_code,stdout,stderr,duration_ms}` where `outcome` is `passed|failed|signaled|spawn_error`; `exit_status` is an integer only for normal process exit, `signal` is a string only for `signaled`, and `error_code` is a stable string only for `spawn_error`.
- `LaneResult` is `{lane,entries,results,exit_status,first_failure}` in manifest order. Its aggregate `exit_status` is the first manifest-ordered nonzero child status, or `1` for a signaled/spawn-error result; it is `0` only when every result passed. `first_failure` retains the exact child status, signal, or spawn error without converting it to success.

```js
import assert from "node:assert/strict";
import test from "node:test";

import {runTestLane} from "../scripts/test-runner.mjs";

test("a lane runs only its explicit stable entries and preserves results",async () => {
  const calls=[];
  const output=[];
  const result=await runTestLane({
    lane:"fast",
    cwd:process.cwd(),
    manifest:{
      schema_version:"toss-test-manifest.v1",concurrency:1,
      lanes:{fast:["test/a.test.js","test/b.test.js"],integration:[],e2e:[],package:[],release:[]},

      calls.push(entry);
      return {
        entry,outcome:"passed",exit_status:0,signal:null,error_code:null,
        stdout:`${entry}\n`,stderr:"",duration_ms:1,
      };
    },
    stdout:{write:value => output.push(value)},stderr:{write() {}},
  });
  assert.deepEqual(calls,["test/a.test.js","test/b.test.js"]);
  assert.deepEqual(result.entries,calls);
  assert.equal(result.exit_status,0);
  assert.match(output.join(""),/lane=fast.*test\/a\.test\.js/s);

test("a failure keeps child output, status, lane, and exact entry",async () => {
  const diagnostics=[];
  const entries=["test/a.test.js","test/b.test.js"];
  const result=await runTestLane({
    lane:"fast",cwd:process.cwd(),eligibleEntries:entries,
    manifest:{
      schema_version:"toss-test-manifest.v1",concurrency:1,
      lanes:{fast:entries,integration:[],e2e:[],package:[],release:[]},

    executeEntry:async entry => entry===entries[0] ? {
      entry,outcome:"passed",exit_status:0,signal:null,error_code:null,
      stdout:"first stdout\n",stderr:"",duration_ms:1,

  assert.equal(result.exit_status,7);
  assert.deepEqual(result.first_failure,{
    entry:"test/b.test.js",outcome:"failed",exit_status:7,
    signal:null,error_code:null,
  });
  assert.match(result.results[1].stdout,/second stdout/);
  assert.match(result.results[1].stderr,/second stderr/);
  assert.match(diagnostics.join(""),/lane=fast.*test\/b\.test\.js.*status=7/s);
});
```
